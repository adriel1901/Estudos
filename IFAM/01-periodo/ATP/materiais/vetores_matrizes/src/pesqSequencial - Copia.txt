#include <stdio.h>

int main(){
	int vet[]={10,5,7,8,9,4,20,45,12,17};
	
	int valorPesq;
	for(int i=0;i<10;i++){

		printf("%d  ",vet[i]);
		printf("testando");
	}
	char continua;
	do{
	   	printf("\n\nDigite um valor para pesquisar ");
		scanf("%d",&valorPesq);  //7
		int conta=0;
		int achou=0;
		while(!achou && conta<10)
		{
			if (valorPesq==vet[conta]){
				achou=1;
			}else{
		   	conta++;	
			}
		}	
		if (achou) {
			printf("Encontrou na posicao %d",conta);
    	} else{
    		printf("Nao existe o numero procurado!");
		}
		
		printf("Pesquisar outro numero? S/N");
		scanf("%c",&continua);
	}while(continua!='N');
	
	return 0;
}
